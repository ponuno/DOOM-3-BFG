# n3dstex: host tool (PC), not part of the 3DS build.
#   cmake -S tools/n3ds/n3dstex -B build-n3dstex && cmake --build build-n3dstex
# On Windows (MinGW / MSVC) n3dstex.exe opens a window when started without arguments.
cmake_minimum_required(VERSION 3.10)
project(n3dstex C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB BINKDEC_SOURCES libbinkdec/src/*.cpp libbinkdec/src/*.c)
set(N3DSTEX_SOURCES n3dstex.cpp n3dssnd.cpp n3dsslim.cpp n3dsvid.cpp rg_etc1.cpp ${BINKDEC_SOURCES})
if(WIN32)
	add_executable(n3dstex WIN32 ${N3DSTEX_SOURCES} n3dsgui_win32.cpp n3dstex.rc)
	target_compile_definitions(n3dstex PRIVATE N3DSTEX_GUI)
	target_link_libraries(n3dstex PRIVATE comctl32 ole32 uuid shell32 advapi32)
	if(MSVC)
		# the manifest comes from n3dstex.rc
		target_link_options(n3dstex PRIVATE /MANIFEST:NO)
	endif()
else()
	add_executable(n3dstex ${N3DSTEX_SOURCES})
endif()
target_include_directories(n3dstex PRIVATE libbinkdec/include)
find_package(Threads REQUIRED)
target_link_libraries(n3dstex PRIVATE Threads::Threads)
if(MINGW)
	target_link_options(n3dstex PRIVATE -static)
endif()
